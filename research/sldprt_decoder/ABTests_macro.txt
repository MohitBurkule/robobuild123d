' ABTests_macro  (SolidWorks 2025 VBA)  - one paste, one F5.
' Builds ~370 small A/B test parts (each = a base + exactly ONE change) in ONE folder you choose
' (use an EMPTY folder), and for each part writes:
'   <part>.SLDPRT   <part>.dump.txt (feature tree, dims, sketches, faces, doc data)   <part>.refs.txt (faces/edges + persistent refs)
' plus manifest.csv and make_log.txt.  Rerun = resume (existing parts are skipped).
' To stop safely: create a file STOP.txt in the folder; the macro ends after the current part.
' Run with NO part open:  Tools > Macro > New > paste over everything > F5.  Then zip the folder and send it.
Option Explicit
Dim swApp As Object
Dim f As Integer          ' dump/refs file
Dim lg As Integer         ' make_log
Dim mf As Integer         ' manifest
Dim gErr As String        ' errors of current variant
Dim gOK As Boolean        ' current variant result
Dim gDir As String
Dim gList As Collection
Dim gN As Long

Const MM As Double = 0.001
Const DEG As Double = 0.0174532925199433

' ======================================================================== main
Sub main()
    Set swApp = Application.SldWorks
    Dim folder As String
    folder = InputBox("Output folder (EMPTY folder; will be created):", "A/B tests", "C:\Users\" & Environ("USERNAME") & "\Desktop\ab_tests\")
    If folder = "" Then Exit Sub
    If Right(folder, 1) <> "\" Then folder = folder & "\"
    On Error Resume Next
    If Dir(folder, vbDirectory) = "" Then MkDir folder
    On Error GoTo 0
    If Dir(folder, vbDirectory) = "" Then MsgBox "Cannot create " & folder: Exit Sub
    gDir = folder
    lg = FreeFile: Open gDir & "make_log.txt" For Append As #lg
    Dim newMf As Boolean: newMf = (Dir(gDir & "manifest.csv") = "")
    mf = FreeFile: Open gDir & "manifest.csv" For Append As #mf
    If newMf Then Print #mf, "file,group,base,change,ok,err"
    Print #lg, "==== start " & Now & " SW " & swApp.RevisionNumber
    swApp.SetUserPreferenceToggle 10, False   ' swInputDimValOnCreate off: no dimension dialog
    Set gList = New Collection
    BuildList
    Print #lg, "variants: " & gList.Count
    Dim k As Variant, t0 As Double: t0 = Timer
    For Each k In gList
        If Dir(gDir & "STOP.txt") <> "" Then Print #lg, "STOP.txt found": Exit For
        RunVariant CStr(k)
        DoEvents
    Next
    Print #lg, "==== end " & Now & " seconds=" & Round(Timer - t0)
    Close #mf: Close #lg
    MsgBox "Done (" & gList.Count & " variants). Zip the folder " & gDir & " and send it."
End Sub

' item = group|desc|base|code|a|b|c
Sub AddV(ByVal grp As String, ByVal desc As String, ByVal base As String, ByVal code As String, Optional ByVal pa As Double = 0, Optional ByVal pb As Double = 0, Optional ByVal pc As Double = 0)
    gN = gN + 1
    gList.Add grp & "|" & desc & "|" & base & "|" & code & "|" & Str(pa) & "|" & Str(pb) & "|" & Str(pc)
End Sub

' ======================================================================== runner
Sub RunVariant(ByVal item As String)
    On Error Resume Next
    Dim p As Variant: p = Split(item, "|")
    Static cnt As Long
    cnt = cnt + 1
    Dim nm As String: nm = "AB_" & p(0) & "_" & Format(cnt, "000") & "_" & p(1)
    Dim path As String: path = gDir & nm & ".SLDPRT"
    If Dir(path) <> "" Then Exit Sub                       ' resume
    gErr = "": gOK = True
    Dim t As Double: t = Timer
    Dim tpl As String: tpl = swApp.GetUserPreferenceStringValue(8)
    Dim doc As Object: Set doc = swApp.NewDocument(tpl, 0, 0, 0)
    If doc Is Nothing Then
        Print #lg, nm & " NO DOC (template '" & tpl & "')"
        Print #mf, nm & ".SLDPRT," & p(0) & "," & p(2) & "," & p(1) & ",False,no template"
        Exit Sub
    End If
    Err.Clear
    Apply doc, CStr(p(3)), Val(p(4)), Val(p(5)), Val(p(6)), nm
    LogErr "apply"
    doc.ClearSelection2 True
    doc.ForceRebuild3 False
    Dim e As Long, w As Long
    doc.Extension.SaveAs3 path, 0, 3, Nothing, Nothing, e, w
    If e <> 0 Then gErr = gErr & " saveErr=" & e: gOK = False
    If Val(p(4)) = 1 And (p(3) = "empty" Or p(3) = "block") Then doc.Save3 1, e, w    ' saved twice
    If p(3) = "lpat" And Val(p(6)) = 8 Then doc.Save3 1, e, w
    LogErr "save"
    DumpAll doc, gDir & nm
    LogErr "dump"
    Dim title As String: title = doc.GetTitle
    Set doc = Nothing
    swApp.CloseDoc title
    If swApp.GetDocumentCount > 0 Then swApp.CloseAllDocuments True
    DoEvents
    Print #lg, nm & " ok=" & gOK & " s=" & Round(Timer - t, 1) & " " & gErr
    Print #mf, nm & ".SLDPRT," & p(0) & "," & p(2) & "," & p(1) & "," & gOK & "," & Replace(Replace(gErr, ",", ";"), vbCrLf, " ")
End Sub

Sub LogErr(ByVal what As String)
    If Err.Number <> 0 Then gErr = gErr & " [" & what & ": " & Err.Number & " " & Err.Description & "]": gOK = False
    Err.Clear
End Sub

Sub Note(ByVal s As String)
    gErr = gErr & " {" & s & "}"
End Sub

Sub Must(ByVal cond As Boolean, ByVal what As String)
    If Not cond Then gOK = False: gErr = gErr & " [" & what & " failed]"
End Sub

Function Sel(ByVal doc As Object, ByVal nm As String, ByVal typ As String, ByVal x As Double, ByVal y As Double, ByVal z As Double, ByVal app As Boolean, ByVal mk As Long) As Boolean
    On Error Resume Next
    Sel = doc.Extension.SelectByID2(nm, typ, x, y, z, app, mk, Nothing, 0)
End Function

Function LastFeat(ByVal doc As Object) As String
    On Error Resume Next
    LastFeat = doc.FeatureByPositionReverse(0).Name
End Function

' ======================================================================== building blocks
' centered rect sketch on Front Plane, returns sketch name
Function RectSketch(ByVal doc As Object, ByVal hw As Double, ByVal hh As Double) As String
    On Error Resume Next
    doc.ClearSelection2 True
    Sel doc, "Front Plane", "PLANE", 0, 0, 0, False, 0
    doc.SketchManager.InsertSketch True
    doc.SketchManager.CreateCenterRectangle 0, 0, 0, hw, hh, 0
    doc.SketchManager.InsertSketch True
    RectSketch = LastFeat(doc)
    LogErr "rectsketch"
End Function

    On Error Resume Next
    Select Case code
        Case "empty"
        Case "sketch": RectSketch doc, 0.01, 0.01

            If b = 0 Then b = 10

        Case "boss": V_Boss doc, a, b
        Case "draft": V_Draft doc, a, b, c
        Case "thin": V_Thin doc, a, b

        Case "cutrad": Block doc, 0.01, 0.01, 0.01: CutF doc, TopCircle(doc, 0, 0, a * MM), True, False, False, 0, 0, 0.005, 0, 0, 0, ""
        Case "fillet": V_Fillet doc, a, b, c
        Case "ffillet", "frfillet": V_FaceFillet doc, code
        Case "chamfer": V_Chamfer doc, a, b, c
        Case "lpat", "lpatprop": V_LPat doc, a, b, c, code
        Case "cpat": V_CPat doc, a, b, c
        Case "mirf": V_MirF doc, a
        Case "mirb": V_MirB doc, a
        Case "patgap": V_PatGap doc, a
        Case "plane": V_Plane doc, a, b
        Case "axis": V_Axis doc, a
        Case "rev": V_Rev doc, a, b
        Case "shell": V_Shell doc, a, b
        Case "loft": V_Loft doc
        Case "sweep": V_Sweep doc
        Case "split": V_Split doc, a
        Case "consumed": V_Consumed doc, a
        Case "skent": V_SkEnt doc, a
        Case "skn": V_SkN doc, a
        Case "rel": V_Rel doc, a
        Case "dim", "dimval", "dimpos": V_Dim doc, code, a
        Case "prop": V_Prop doc, a
        Case "mat": V_Mat doc, a
        Case "cfg": V_Cfg doc, a
        Case "eqn": V_Eqn doc, a
        Case "colour": V_Colour doc, a
        Case "units": V_Units doc, a
        Case "view": V_View doc, a
        Case "ops": V_Ops doc, a
        Case "selset": V_SelSet doc, a
        Case "fav": V_Fav doc, a
        Case Else: Note "unknown code " & code: gOK = False
    End Select
    LogErr "dispatch " & code
End Sub

Function LinPattern(ByVal doc As Object, ByVal n As Long, ByVal sp As Double, ByVal n2 As Long, ByVal flip As Boolean, ByVal seeds As String, ByVal onlyOld As Boolean) As Object
    On Error Resume Next
    Dim tryN As Integer, ft As Object, s As Variant, ok As Boolean
    For tryN = 1 To 5
        If onlyOld And tryN < 5 Then GoTo nxt
        doc.ClearSelection2 True
        ok = Sel(doc, "", "EDGE", 0, -0.01, 0, False, 1)
        If n2 > 1 Then ok = Sel(doc, "", "EDGE", -0.01, 0, 0, True, 2) And ok
        For Each s In Split(seeds, ";")
            ok = Sel(doc, CStr(s), "BODYFEATURE", 0, 0, 0, True, 4) And ok
        Next
        Set ft = Nothing
        Dim n2v As Long: n2v = IIf(n2 > 1, n2, 1)
        Select Case tryN
            Case 1: Set ft = doc.FeatureManager.FeatureLinearPattern5(n, sp, n2v, 0.006, flip, False, "", "", False, False, False, False, False, False, False, False, False, False, 0, 0, False, False)
            Case 2: Set ft = doc.FeatureManager.FeatureLinearPattern4(n, sp, n2v, 0.006, flip, False, "", "", False, False, False, False, False, False, False, 0, 0, False, False)
            Case 3: Set ft = doc.FeatureManager.FeatureLinearPattern3(n, sp, n2v, 0.006, flip, False, "", "", False, False)
            Case 4: Set ft = doc.FeatureManager.FeatureLinearPattern2(n, sp, n2v, 0.006, flip, False, "", "", False)
            Case 5
                doc.FeatureLinearPattern n, sp, n2v, 0.006, flip, False, "", ""
                If LastFeat(doc) Like "LPattern*" Then Set ft = doc.FeatureByPositionReverse(0)
        End Select
        LogErr "linpattern try " & tryN & " sel=" & ok
        If Not ft Is Nothing Then
            If LastFeat(doc) Like "LPattern*" Then Note "lpattern by try " & tryN: Exit For
            Note "lpattern try " & tryN & " returned object but last feature=" & LastFeat(doc)
            Set ft = Nothing
        End If
nxt:
    Next
    Set LinPattern = ft
    Must Not ft Is Nothing, "linear pattern"
End Function

Sub V_Ops(ByVal doc As Object, ByVal k As Double)
    On Error Resume Next
    ' base "chamf": block + fillet1 on back edge + chamfer1 on left edge (like T04/T05)
    Block doc, 0.01, 0.01, 0.01
    doc.ClearSelection2 True: SelEdge doc, 0, False, 1: Fillet doc, 0.001, 195, 0
    doc.ClearSelection2 True: SelEdge doc, 3, False, 1: Chamfer doc, 1, 0.001, 45 * DEG, 0, 0, 0, 0, 0
    doc.ClearSelection2 True

        Case 1: doc.FeatureByName("Fillet1").Name = "MyFillet"
        Case 2: doc.FeatureByName("Sketch1").Name = "MySketch"
        Case 3
            Dim bs As Variant: bs = doc.GetBodies2(0, False)
            bs(0).Name = "MyBody"
        Case 4: Sel doc, "Chamfer1", "BODYFEATURE", 0, 0, 0, False, 0: doc.EditSuppress2
        Case 5: Sel doc, "Fillet1", "BODYFEATURE", 0, 0, 0, False, 0: doc.EditSuppress2
        Case 6
            Sel doc, "Chamfer1", "BODYFEATURE", 0, 0, 0, False, 0: doc.EditSuppress2
            doc.ClearSelection2 True
            Sel doc, "Chamfer1", "BODYFEATURE", 0, 0, 0, False, 0: doc.EditUnsuppress2
        Case 7: Must doc.Extension.ReorderFeature("Chamfer1", "Fillet1", 2), "ReorderFeature"   ' swMoveLocation_e 2 = before?
        Case 8: Sel doc, "Chamfer1", "BODYFEATURE", 0, 0, 0, False, 0: doc.Extension.DeleteSelection2 0
        Case 9: Must doc.FeatureManager.EditRollback(2, "Chamfer1"), "EditRollback"
        Case 10
            Dim b2 As Variant: b2 = doc.GetBodies2(0, False)
            b2(0).HideBody True
        Case 11: Sel doc, "Sketch1", "SKETCH", 0, 0, 0, False, 0: doc.BlankSketch: doc.ClearSelection2 True: Sel doc, "Sketch1", "SKETCH", 0, 0, 0, False, 0: doc.UnblankSketch
        Case 12: Set ft = doc.FeatureByName("Fillet1"): ft.Description = "my comment": doc.Extension.AddComment "my comment"
        Case 13
            Sel doc, "Fillet1", "BODYFEATURE", 0, 0, 0, False, 0
            Sel doc, "Chamfer1", "BODYFEATURE", 0, 0, 0, True, 0
            doc.FeatureManager.InsertFeatureTreeFolder2 1    ' swFeatureTreeFolder_Containing
    End Select

Sub DumpAll(ByVal doc As Object, ByVal base As String)
    On Error Resume Next
    f = FreeFile: Open base & ".dump.txt" For Output As #f
    DumpFeatures doc
    DumpDoc doc
    Close #f
    Err.Clear
    f = FreeFile: Open base & ".refs.txt" For Output As #f
    DumpRefs doc
    Close #f
    Err.Clear
End Sub

Sub DumpFeatures(ByVal doc As Object)
    On Error Resume Next
    Print #f, "PART " & doc.GetTitle
    Dim feat As Object: Set feat = doc.FirstFeature
    Dim idx As Long
    Do While Not feat Is Nothing
        DumpFeature doc, feat, idx, 0: idx = idx + 1
        Dim s As Object: Set s = feat.GetFirstSubFeature
        Do While Not s Is Nothing
            DumpFeature doc, s, -1, 1
            Set s = s.GetNextSubFeature
        Loop
        Set feat = feat.GetNextFeature
    Loop
    Set feat = Nothing: Set s = Nothing
End Sub

Sub DumpFeature(ByVal doc As Object, ByVal feat As Object, ByVal idx As Long, ByVal lvl As Integer)
    On Error Resume Next
    Dim t As String: t = feat.GetTypeName2
    Print #f, String(lvl * 2, " ") & "FEATURE " & idx & " | " & feat.Name & " | " & t & " | id=" & feat.GetID & " | suppressed=" & feat.IsSuppressed
    Dim d As Object: Set d = feat.GetFirstDisplayDimension
    Do While Not d Is Nothing
        Dim dm As Object: Set dm = d.GetDimension2(0)
        Print #f, "  DIM " & dm.FullName & " = " & dm.SystemValue & " driven=" & dm.DrivenState & " type=" & d.Type2
        Set d = feat.GetNextDisplayDimension(d)
    Loop
    Dim par As Variant: par = feat.GetParents
    Dim i As Long
    If Not IsEmpty(par) Then
        For i = 0 To UBound(par): Print #f, "  PARENT " & par(i).Name & " | " & par(i).GetTypeName2: Next
    End If
    Dim fs As Variant: fs = feat.GetFaces
    If Not IsEmpty(fs) Then
        For i = 0 To UBound(fs): Print #f, "  FACE " & RefDesc(fs(i)) & " area=" & fs(i).GetArea: Next
    End If
    Select Case t
        Case "ProfileFeature", "3DProfileFeature": DumpSketch feat.GetSpecificFeature2
        Case "Extrusion", "Cut", "Boss", "BossThin", "CutThin", "ICE", "Extrusion Thin": DumpExtrude doc, feat
        Case "Fillet", "Chamfer", "Chamfer2": DumpFillet doc, feat, t
        Case "RefPlane": DumpPlane feat
        Case "LPattern", "CirPattern", "MirrorPattern", "MirrorSolid", "SketchPattern": DumpPattern doc, feat, t
        Case "Revolution", "RevCut", "RevolutionThin", "RevCutThin": DumpRev feat
        Case "Shell": Print #f, "  SHELL t=" & feat.GetDefinition.Thickness & " out=" & feat.GetDefinition.ShellOutward
    End Select
    Err.Clear
End Sub

Sub DumpSketch(ByVal sk As Object)
    On Error Resume Next
    Dim a As Variant: a = sk.ModelToSketchTransform.Inverse.ArrayData
    Print #f, "  SKETCH_TO_MODEL " & JoinV(a)
    Dim pts As Variant: pts = sk.GetSketchPoints2
    Dim i As Long, id As Variant
    If Not IsEmpty(pts) Then
        For i = 0 To UBound(pts)
            id = pts(i).GetID
            Print #f, "  PT " & id(0) & "_" & id(1) & " (" & pts(i).X & "," & pts(i).Y & "," & pts(i).Z & ")"
        Next
    End If
    Dim segs As Variant: segs = sk.GetSketchSegments
    If Not IsEmpty(segs) Then
        For i = 0 To UBound(segs)
            Dim s As Object: Set s = segs(i)
            id = s.GetID
            Dim ln As String: ln = "  SEG " & id(0) & "_" & id(1) & " type=" & s.GetType & " constr=" & s.ConstructionGeometry
            Select Case s.GetType
                Case 0: ln = ln & " p0=" & PP(s.GetStartPoint2) & " p1=" & PP(s.GetEndPoint2)
                Case 1: ln = ln & " c=" & PP(s.GetCenterPoint2) & " p0=" & PP(s.GetStartPoint2) & " p1=" & PP(s.GetEndPoint2) & " r=" & s.GetRadius & " dir=" & s.GetRotationDir
                Case Else: ln = ln & " len=" & s.GetLength
            End Select
            Print #f, ln
        Next
    End If
    Dim rels As Variant: rels = sk.RelationManager.GetRelations(0)
    Dim es As String, k As Long, ents As Variant, eid As Variant
    If Not IsEmpty(rels) Then
        For i = 0 To UBound(rels)
            ents = rels(i).GetEntities: es = ""
            If Not IsEmpty(ents) Then
                For k = 0 To UBound(ents): eid = ents(k).GetID: es = es & eid(0) & "_" & eid(1) & " ": Next
            End If
            Print #f, "  REL type=" & rels(i).GetRelationType & " ents=" & es
        Next
    End If
    Dim cs As Variant: cs = sk.GetSketchContours
    If Not IsEmpty(cs) Then
        For i = 0 To UBound(cs)
            Dim cseg As Variant: cseg = cs(i).GetSketchSegments
            es = ""
            For k = 0 To UBound(cseg): eid = cseg(k).GetID: es = es & eid(0) & "_" & eid(1) & " ": Next
            Print #f, "  CONTOUR " & i & " closed=" & cs(i).IsClosed & " segs=" & es
        Next
    End If
End Sub

Sub DumpDoc(ByVal doc As Object)
    On Error Resume Next
    Dim i As Long, nms As Variant, vv As Variant, vt As Variant, rv As Variant
    Print #f, "DOC"
    nms = doc.GetConfigurationNames
    For i = 0 To UBound(nms): Print #f, "  CONFIG " & nms(i): Next
    Print #f, "  ACTIVE_CONFIG " & doc.ConfigurationManager.ActiveConfiguration.Name
    Dim ds As Variant: ds = doc.ConfigurationManager.ActiveConfiguration.GetDisplayStates
    For i = 0 To UBound(ds): Print #f, "  DISPLAY_STATE " & ds(i): Next
    Dim cp As Object: Set cp = doc.Extension.CustomPropertyManager("")
    Dim n As Long: n = cp.GetAll3(nms, vt, vv, rv, Empty)
    For i = 0 To n - 1: Print #f, "  PROP " & nms(i) & " type=" & vt(i) & " = " & vv(i): Next
    Dim db As String: Print #f, "  MATERIAL " & doc.GetMaterialPropertyName2("", db) & " db=" & db
    Print #f, "  COLOUR " & JoinV(doc.MaterialPropertyValues)
    Dim em As Object: Set em = doc.GetEquationMgr
    For i = 0 To em.GetCount - 1: Print #f, "  EQN " & em.Equation(i) & " = " & em.Value(i): Next
    Print #f, "  UNITS linear=" & doc.GetUserPreferenceIntegerValue(51) & " dec=" & doc.GetUserPreferenceIntegerValue(52)
    Dim bs As Variant: bs = doc.GetBodies2(0, False)
    If Not IsEmpty(bs) Then
        For i = 0 To UBound(bs): Print #f, "  BODY " & bs(i).Name & " faces=" & bs(i).GetFaceCount & " edges=" & bs(i).GetEdgeCount & " visible=" & bs(i).Visible: Next
    End If
    Print #f, "  FEATURECOUNT " & doc.FeatureManager.GetFeatureCount(True)
End Sub

Sub DumpRefs(ByVal doc As Object)
    On Error Resume Next
    Print #f, "PART " & doc.GetTitle
    Dim ext As Object: Set ext = doc.Extension
    Dim feat As Object: Set feat = doc.FirstFeature
    Dim fs As Variant, es As Variant, i As Long, j As Long, t As String
    Do While Not feat Is Nothing
        t = feat.GetTypeName2
        Print #f, "FEATURE " & feat.Name & " | " & t & " | id=" & feat.GetID & " | pref=" & PRef(ext, feat)
        fs = feat.GetFaces
        If Not IsEmpty(fs) Then
            For i = 0 To UBound(fs)
                Print #f, "  FACE " & i & " fid=" & fs(i).GetFaceId & " area=" & fs(i).GetArea & " " & SurfDesc(fs(i)) & " pref=" & PRef(ext, fs(i))
                es = fs(i).GetEdges
                If Not IsEmpty(es) Then
                    For j = 0 To UBound(es): Print #f, "    EDGE " & EdgeDesc(es(j)) & " pref=" & PRef(ext, es(j)): Next
                End If
            Next
        End If
        If t = "Fillet" Or t = "Chamfer" Then DumpBlendEdges doc, ext, feat
        Set feat = feat.GetNextFeature
    Loop
    Dim bodies As Variant: bodies = doc.GetBodies2(0, False)
    Dim be As Variant, b As Long
    If Not IsEmpty(bodies) Then
        For b = 0 To UBound(bodies)
            Print #f, "BODY " & b & " " & bodies(b).Name
            be = bodies(b).GetEdges
            If Not IsEmpty(be) Then
                For i = 0 To UBound(be): Print #f, "  BEDGE " & EdgeDesc(be(i)) & " pref=" & PRef(ext, be(i)): Next
            End If
        Next
    End If
    Set feat = Nothing: Set ext = Nothing: fs = Empty: es = Empty: bodies = Empty: be = Empty
End Sub

Sub DumpBlendEdges(ByVal doc As Object, ByVal ext As Object, ByVal feat As Object)
    On Error Resume Next
    Dim x As Object: Set x = feat.GetDefinition
    x.AccessSelections doc, Nothing
    Dim n As Long, i As Long, it As Object, tp As Long
    If feat.GetTypeName2 = "Fillet" Then n = x.FilletItemsCount Else n = x.GetEdgeCount
    Print #f, "  BLEND_ITEMS " & n
    For i = 0 To n - 1
        If feat.GetTypeName2 = "Fillet" Then Set it = x.GetFilletItemAtIndex2(i, tp) Else Set it = x.GetEdge(i)
        If Not it Is Nothing Then Print #f, "  BLEND_EDGE " & i & " " & EdgeDesc(it) & " pref=" & PRef(ext, it)
    Next
    x.ReleaseSelectionAccess
End Sub

Function EdgeDesc(ByVal e As Object) As String
    On Error Resume Next
    Dim cp As Object: Set cp = e.GetCurveParams3
    Dim c As Object: Set c = e.GetCurve
    Dim k As String
    If c.IsLine Then
        k = "line"
    ElseIf c.IsCircle Then
        k = "circle " & JoinV(c.CircleParams)
    Else
        k = "curve"
    End If
    EdgeDesc = k & " s=(" & cp.StartPoint(0) & "," & cp.StartPoint(1) & "," & cp.StartPoint(2) & ") e=(" & cp.EndPoint(0) & "," & cp.EndPoint(1) & "," & cp.EndPoint(2) & ") id=" & e.GetID
End Function

Function SurfDesc(ByVal o As Object) As String
    On Error Resume Next
    Dim sf As Object: Set sf = o.GetSurface
    If sf.IsPlane Then
        SurfDesc = "plane " & JoinV(sf.PlaneParams)
    ElseIf sf.IsCylinder Then
        SurfDesc = "cyl " & JoinV(sf.CylinderParams)
    ElseIf sf.IsCone Then
        SurfDesc = "cone " & JoinV(sf.ConeParams)
    ElseIf sf.IsTorus Then
        SurfDesc = "torus " & JoinV(sf.TorusParams)
    Else
        SurfDesc = "surf " & sf.Identity
    End If
    SurfDesc = SurfDesc & " box=" & JoinV(o.GetBox)
End Function

Function RefDesc(ByVal o As Object) As String
    On Error Resume Next
    Dim sf As Object: Set sf = o.GetSurface
    If sf Is Nothing Then RefDesc = TypeName(o): Exit Function
    Dim p As Variant
    If sf.IsPlane Then
        p = sf.PlaneParams: RefDesc = "plane n=(" & p(0) & "," & p(1) & "," & p(2) & ") o=(" & p(3) & "," & p(4) & "," & p(5) & ")"
    ElseIf sf.IsCylinder Then
        p = sf.CylinderParams: RefDesc = "cyl o=(" & p(0) & "," & p(1) & "," & p(2) & ") ax=(" & p(3) & "," & p(4) & "," & p(5) & ") r=" & p(6)
    ElseIf sf.IsCone Then
        RefDesc = "cone " & JoinV(sf.ConeParams)
    ElseIf sf.IsTorus Then
        RefDesc = "torus " & JoinV(sf.TorusParams)
    ElseIf sf.IsSphere Then
        RefDesc = "sphere " & JoinV(sf.SphereParams)
    Else
        RefDesc = "surf id=" & sf.Identity
    End If
    Dim bb As Variant: bb = o.GetBox
    If Not IsEmpty(bb) Then RefDesc = RefDesc & " box=" & JoinV(bb)
End Function

Function PRef(ByVal ext As Object, ByVal o As Object) As String
    On Error Resume Next
    Dim v As Variant: v = ext.GetPersistReference3(o)
    If IsEmpty(v) Then PRef = "-": Exit Function
    Dim i As Long, s As String
    For i = LBound(v) To UBound(v): s = s & Right("0" & Hex(v(i)), 2): Next
    PRef = s
End Function

Function PP(ByVal v As Object) As String
    On Error Resume Next
    PP = "(" & v.X & "," & v.Y & "," & v.Z & ")"
End Function

Function JoinV(ByVal a As Variant) As String
    On Error Resume Next
    Dim i As Long, s As String
    For i = LBound(a) To UBound(a): s = s & a(i) & ",": Next
    JoinV = s
End Function
